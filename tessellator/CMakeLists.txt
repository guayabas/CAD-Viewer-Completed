cmake_minimum_required(VERSION 3.20)
project(cad_viewer_tessellator CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(OpenCASCADE CONFIG REQUIRED)
add_executable(tessellator tessellator.cpp face_map_writer.cpp)
target_compile_options(tessellator PRIVATE -Wall -Wextra)
# SYSTEM hides warnings from OCCT headers; -Wall -Wextra still apply to our code.
target_include_directories(tessellator SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(tessellator PRIVATE TKXDESTEP TKSTEP TKMesh TKTopAlgo TKXCAF TKLCAF TKRWMesh)
